def make_set (db_cx, table, field, where):
    """ Build a set and return it from a table/field"""
    cu = db_cx.cursor()
    cu.execute("SELECT "+field+" FROM "+table+" "+where+";")
    s = set()
    for row in cu.fetchall():
        s.add(row[field])
    return s
