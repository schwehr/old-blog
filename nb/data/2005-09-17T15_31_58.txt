WARNING: I released this script to the world a little early.  It does not make use of the timeStart and timeEnd.

I am not quite up to writing another program to generate pltsegy files
with best guess parameters.  Instead, I wrote a short bash shell
script.  This is easier now and I will pay for it later.  I tried to
use the sqlite database from each xstar file as much as possible.  I
got thrown off by the 41 coming back in the SampleInterval field.
This is really the value stored in the xstar SampleInterval field
(bytes 58 and 59), but not the value to use.  The actual
SampleInterval is 83 (0.000083) which sioseis sets explicitly with a
special case when it sees ibuf(59) .EQ. 41.  Then it also sets buf(49)
to 1/12000 (0.00008333), but I have not bothered to look up which
field that is.  ibuf is and INTEGER*2, and I am not sure what type buf
is.
<pre>
<font color="0000ff"><strong>#!/bin/bash</strong></font>
<font color="#FF8800"># Why am I writing this in bash???</font>
declare -ri VERBOSITY=2
<font color="#2040a0">project</font>=<font color="#008000">&quot;Eel_River_Basin&quot;</font>
<font color="#2040a0">name</font>=EEL5kL01-00001
<font color="#2040a0">db</font>=<font color="#008000">&quot;${name}.xstar.db&quot;</font>
if [ ! -f $db ]; then
    echo &quot;ERROR: missing database ($db)&quot;
    exit
fi
<font color="#2040a0">minShot</font>=`sqlite <font color="#2040a0">$db</font> 'select min(Shotpoint) from segyTrace'`
<font color="#2040a0">maxShot</font>=`sqlite <font color="#2040a0">$db</font> 'select max(Shotpoint) from segyTrace'`

echo shotrange = <font color="#2040a0">$minShot</font> ... <font color="#2040a0">$maxShot</font>

<font color="#2040a0">minDelay</font>=`sqlite <font color="#2040a0">$db</font> 'SELECT min(Delay) FROM segyTrace'`
<font color="#2040a0">maxDelay</font>=`sqlite <font color="#2040a0">$db</font> 'SELECT max(Delay) FROM segyTrace'`

<font color="#FF8800"># FIX: make sure min/max are the same</font>
<font color="#2040a0">TraceSamples</font>=`sqlite <font color="#2040a0">$db</font> 'SELECT min(TraceSamples) FROM segyTrace'`

<font color="#FF8800">#SampleInterval=`sqlite  $db 'select max(sampleinterval) from segyTrace'`</font>
<font color="#FF8800"># 41 that is in the db file is NOT ok.  It is 83</font>
SampleInterval=0.000083

<font color="#2040a0">timeStart</font>=`python -c <font color="#008000">&quot;print \&quot;%0.2f\&quot; % (<font color="#2040a0">$minDelay</font>/1000.)&quot;</font>`
<font color="#2040a0">timeEnd</font>=`python -c <font color="#008000">&quot;print (<font color="#2040a0">$minDelay</font>/1000. + <font color="#2040a0">$SampleInterval</font> * <font color="#2040a0">$TraceSamples</font>)&quot;</font>`

echo <font color="#2040a0">$minDelay</font>  <font color="#2040a0">$SampleInterval</font> <font color="#2040a0">$TraceSamples</font>

echo <font color="#008000">&quot;Time range: <font color="#2040a0">$timeStart</font> <font color="#2040a0">$timeEnd</font>&quot;</font>

<font color="#2040a0">pltWidth</font>=350<font color="#FF8800"> # The default</font>

<font color="#FF8800"># Screen width / number of shots we want to see</font>
<font color="#2040a0">spa</font>=`python -c <font color="#008000">&quot;print \&quot;%.2f\&quot; % (<font color="#2040a0">$pltWidth</font>/float(<font color="#2040a0">$maxShot</font>-<font color="#2040a0">$minShot</font>))&quot;</font>`
echo <font color="#008000">&quot;Spacing = <font color="#2040a0">$spa</font>&quot;</font>

<font color="#FF8800"># SPA of 0.5 showed some</font>
makepltsegy.py --verbosity=$VERBOSITY -o EEL5kL01-00001.pltsegy -i $name.sgy \
  -s <font color="#2040a0">$minShot</font> -S <font color="#2040a0">$maxShot</font> \
  --<font color="#2040a0">spa</font>=<font color="#2040a0">$spa</font> --<font color="#2040a0">timescale</font>=200 \
  --<font color="#2040a0">title</font>=<font color="#008000">&quot;<font color="#2040a0">$project</font> <font color="#2040a0">$name</font> <font color="#2040a0">$minShot</font> ... <font color="#2040a0">$maxShot</font>&quot;</font>
