Here is a very basic sioseis script.  When I am working on rendering a
small upset of the line, it is nice to have just a small subset of the
shots in a segy file.  This makes the rest of the process quicker.  I
then don't have to worry about first and last shots and disk usage is
kept to a minimum.  First use lsd to see the total number of shots.  My subset will be from shots 88500 to 90000
<pre>

<font color="0000ff"><strong>#!/bin/bash</strong></font>
declare -ri <font color="#2040a0">EXIT_FAILURE</font>=1
if [ <font color="#2040a0">$#</font> != 3 ] ; then
  echo
  echo <font color="#008000">&quot;  ERROR: Must specify 3 arguments.  Found <font color="#2040a0">$#</font>&quot;</font>
  echo
  echo <font color="#008000">&quot;  usage:  <font color="#2040a0">$0</font> [segy in file] [start] [end] &quot;</font> 
  echo
  echo <font color="#008000">&quot;  Example:&quot;</font>
  echo <font color="#008000">&quot;    <font color="#2040a0">$0</font> longline.sgy 88500 90000&quot;</font>
  exit <font color="#2040a0">$EXIT_FAILURE</font>
fi
declare -ri <font color="#2040a0">start</font>=<font color="#2040a0">$2</font>
declare -ir <font color="#2040a0">end</font>=<font color="#2040a0">$3</font>
declare -r <font color="#2040a0">INFILE</font>=<font color="#2040a0">$1</font>
declare -r <font color="#2040a0">OUTFILE</font>=${INFILE%%.sgy}-<font color="#2040a0">$start</font>-<font color="#2040a0">$end</font>.sgy
echo RANGE:  <font color="#2040a0">$start</font> ... <font color="#2040a0">$end</font>
echo INFILE <font color="#2040a0">$INFILE</font>
echo OUTFILE <font color="#2040a0">$OUTFILE</font>
<font color="#2040a0">OS</font>=`uname -s`
<font color="#FF8800">#sioseis=${HOME}/bin/sioseis.${OS}</font>
sioseis=sioseis
$sioseis &lt;&lt; EOF 
<em>procs diskin diskoa end
diskin
  ipath <font color="#2040a0">$INFILE</font> end
end
!
diskoa
 opath <font color="#2040a0">$OUTFILE</font>  fno <font color="#2040a0">$start</font> lno <font color="#2040a0">$end</font> end
end
end</em>
EOF
</pre>
