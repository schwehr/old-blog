The USGS has a handy name database full of all sorts of strange names.
Here is some python code to quickly parse the name, type of name, and
location.  This first program (gnis.py) creates a sqlite database.
<pre>
<font color="#444444">#!/usr/bin/env python</font>

<font color="#008000">&quot;&quot;</font><font color="#008000">&quot;
Parse the USGS Geophraphic Names Information System (GNIS) files and
create an sqlite database.

http://geonames.usgs.gov/stategaz/
&quot;</font><font color="#008000">&quot;&quot;</font>
<strong>import</strong> <font color="#2040a0">sqlite</font>
<strong>import</strong> <font color="#2040a0">os</font>
<strong>import</strong> <font color="#2040a0">sys</font>

<strong>if</strong> <font color="#2040a0">__name__</font> <font color="4444FF">=</font><font color="4444FF">=</font> <font color="#008000">&quot;__main__&quot;</font><font color="4444FF">:</font>
    <font color="#2040a0">cx</font> <font color="4444FF">=</font> <font color="#2040a0">sqlite</font>.<font color="#2040a0">connect</font><font color="4444FF">(</font><font color="#008000">'ca-names.db'</font><font color="4444FF">)</font> <font color="#444444"># No, this has nothing to do with DNS!!</font>
    <font color="#2040a0">cu</font> <font color="4444FF">=</font> <font color="#2040a0">cx</font>.<font color="#2040a0">cursor</font><font color="4444FF">(</font><font color="4444FF">)</font>
    <font color="#2040a0">cu</font>.<font color="#2040a0">execute</font><font color="4444FF">(</font><font color="#008000">'CREATE TABLE places ( name VARCHAR(60), type VARCHAR(50), lon REAL, lat REAL);'</font><font color="4444FF">)</font>
    <font color="#2040a0">cx</font>.<font color="#2040a0">commit</font><font color="4444FF">(</font><font color="4444FF">)</font>
    <font color="#2040a0">count</font> <font color="4444FF">=</font> <font color="#FF0000">0</font>
    <strong>for</strong> <font color="#2040a0">line</font> <strong>in</strong> <font color="#2040a0">open</font> <font color="4444FF">(</font><font color="#008000">'CA_DECI.TXT'</font>,<font color="#008000">'r'</font><font color="4444FF">)</font>.<font color="#2040a0">xreadlines</font><font color="4444FF">(</font><font color="4444FF">)</font><font color="4444FF">:</font>
        <font color="#2040a0">count</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#FF0000">1</font>
        <strong>if</strong> <font color="#2040a0">count</font><font color="4444FF">%</font><font color="#FF0000">500</font><font color="4444FF">=</font><font color="4444FF">=</font><font color="#FF0000">0</font><font color="4444FF">:</font> <strong>print</strong> <font color="#2040a0">count</font>
        
        <font color="#2040a0">fields</font> <font color="4444FF">=</font> <font color="#2040a0">line</font>.<font color="#2040a0">split</font><font color="4444FF">(</font><font color="#008000">'|'</font><font color="4444FF">)</font>
        <font color="#2040a0">name</font> <font color="4444FF">=</font> <font color="#2040a0">fields</font><font color="4444FF">[</font><font color="#FF0000">2</font><font color="4444FF">]</font>
        <strong>if</strong> <font color="4444FF">-</font><font color="#FF0000">1</font> <font color="4444FF">!</font><font color="4444FF">=</font> <font color="#2040a0">name</font>.<font color="#2040a0">find</font><font color="4444FF">(</font><font color="#008000">&quot;'&quot;</font><font color="4444FF">)</font><font color="4444FF">:</font>
            <strong>print</strong> <font color="#008000">'skipping '</font>,<font color="#2040a0">name</font>,
            <strong>continue</strong>
        <font color="#2040a0">type</font> <font color="4444FF">=</font> <font color="#2040a0">fields</font><font color="4444FF">[</font><font color="#FF0000">3</font><font color="4444FF">]</font>
        <strong>try</strong><font color="4444FF">:</font>
            <font color="#2040a0">lon</font> <font color="4444FF">=</font> <font color="#2040a0">float</font><font color="4444FF">(</font><font color="#2040a0">fields</font><font color="4444FF">[</font><font color="#FF0000">10</font><font color="4444FF">]</font><font color="4444FF">)</font>
            <font color="#2040a0">lat</font> <font color="4444FF">=</font> <font color="#2040a0">float</font><font color="4444FF">(</font><font color="#2040a0">fields</font><font color="4444FF">[</font><font color="#FF0000">9</font><font color="4444FF">]</font><font color="4444FF">)</font>
        <strong>except</strong> <font color="#2040a0">ValueError</font><font color="4444FF">:</font>
            <strong>print</strong> <font color="#008000">&quot;skipping line (bad lon/lat): &quot;</font>,<font color="#2040a0">line</font>,
            <strong>continue</strong>
        <font color="#2040a0">insertStr</font>  <font color="4444FF">=</font> <font color="#008000">'INSERT INTO places (name,type,lon,lat) VALUES ('</font>
        <font color="#2040a0">insertStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">&quot;'&quot;</font><font color="4444FF">+</font><font color="#2040a0">name</font><font color="4444FF">+</font><font color="#008000">&quot;'&quot;</font>
        <font color="#2040a0">insertStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">&quot;, '&quot;</font><font color="4444FF">+</font><font color="#2040a0">type</font><font color="4444FF">+</font><font color="#008000">&quot;'&quot;</font>
        <font color="#2040a0">insertStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">', '</font><font color="4444FF">+</font><font color="#2040a0">str</font><font color="4444FF">(</font><font color="#2040a0">lon</font><font color="4444FF">)</font>
        <font color="#2040a0">insertStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">', '</font><font color="4444FF">+</font><font color="#2040a0">str</font><font color="4444FF">(</font><font color="#2040a0">lat</font><font color="4444FF">)</font>
        <font color="#2040a0">insertStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">');'</font>
        <font color="#2040a0">cu</font>.<font color="#2040a0">execute</font> <font color="4444FF">(</font><font color="#2040a0">insertStr</font><font color="4444FF">)</font>
    <font color="#2040a0">cx</font>.<font color="#2040a0">commit</font><font color="4444FF">(</font><font color="4444FF">)</font>

</pre>
Then I wrote up something to search a bounding box and look for names
that are of type cape, channel, or stream.  The I only wanted rivers,
so toss out a few extra names.  
<pre>
<font color="#444444">#!/usr/bin/env python</font>
<font color="#008000">&quot;&quot;</font><font color="#008000">&quot;
Get the Eel River Basin names
&quot;</font><font color="#008000">&quot;&quot;</font>
<strong>import</strong> <font color="#2040a0">sqlite</font>
<strong>import</strong> <font color="#2040a0">os</font>
<strong>import</strong> <font color="#2040a0">sys</font>

<strong>if</strong> <font color="#2040a0">__name__</font> <font color="4444FF">=</font><font color="4444FF">=</font> <font color="#008000">&quot;__main__&quot;</font><font color="4444FF">:</font>
    <font color="#2040a0">cx</font> <font color="4444FF">=</font> <font color="#2040a0">sqlite</font>.<font color="#2040a0">connect</font><font color="4444FF">(</font><font color="#008000">'ca-names.db'</font><font color="4444FF">)</font> <font color="#444444"># No, this has nothing to do with DNS!!</font>
    <font color="#2040a0">cu</font> <font color="4444FF">=</font> <font color="#2040a0">cx</font>.<font color="#2040a0">cursor</font><font color="4444FF">(</font><font color="4444FF">)</font>
    <font color="#2040a0">searchStr</font>  <font color="4444FF">=</font> <font color="#008000">'SELECT lon,lat,name,type FROM places WHERE '</font>
    <font color="#2040a0">searchStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">' lon &gt; -125.0'</font>
    <font color="#2040a0">searchStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">' AND lon &lt; -123.8'</font> <font color="#444444"># -123.0'</font>
    <font color="#2040a0">searchStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">' AND lat &gt; 40.1'</font>   <font color="#444444"># 39.5'</font>
    <font color="#2040a0">searchStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">' AND lat &lt; 42'</font>
    <font color="#2040a0">searchStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">' AND (type=&quot;cape&quot; OR type=&quot;channel&quot; OR type=&quot;stream&quot;)'</font>
    <font color="#2040a0">searchStr</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#008000">';'</font>
    <font color="#2040a0">cu</font>.<font color="#2040a0">execute</font><font color="4444FF">(</font><font color="#2040a0">searchStr</font><font color="4444FF">)</font>
    <strong>for</strong> <font color="#2040a0">row</font> <strong>in</strong> <font color="#2040a0">cu</font>.<font color="#2040a0">fetchall</font><font color="4444FF">(</font><font color="4444FF">)</font><font color="4444FF">:</font>
        <strong>if</strong> <font color="4444FF">(</font><font color="4444FF">-</font><font color="#FF0000">1</font> <font color="4444FF">!</font><font color="4444FF">=</font> <font color="#2040a0">row</font><font color="4444FF">[</font><font color="#008000">'name'</font><font color="4444FF">]</font>.<font color="#2040a0">find</font><font color="4444FF">(</font><font color="#008000">'Stream'</font><font color="4444FF">)</font><font color="4444FF">)</font> <strong>or</strong> \
           <font color="4444FF">-</font><font color="#FF0000">1</font> <font color="4444FF">!</font><font color="4444FF">=</font> <font color="#2040a0">row</font><font color="4444FF">[</font><font color="#008000">'name'</font><font color="4444FF">]</font>.<font color="#2040a0">find</font><font color="4444FF">(</font><font color="#008000">'Creek'</font><font color="4444FF">)</font> <strong>or</strong> \
           <font color="4444FF">-</font><font color="#FF0000">1</font> <font color="4444FF">!</font><font color="4444FF">=</font> <font color="#2040a0">row</font><font color="4444FF">[</font><font color="#008000">'name'</font><font color="4444FF">]</font>.<font color="#2040a0">find</font><font color="4444FF">(</font><font color="#008000">'Slough'</font><font color="4444FF">)</font><font color="4444FF">:</font>
            <strong>print</strong> <font color="#008000">'skipping '</font>, <font color="#2040a0">row</font><font color="4444FF">[</font><font color="#008000">'name'</font><font color="4444FF">]</font>
            <strong>continue</strong>
        <strong>print</strong> <font color="#2040a0">row</font><font color="4444FF">[</font><font color="#008000">'lon'</font><font color="4444FF">]</font>, <font color="#2040a0">row</font><font color="4444FF">[</font><font color="#008000">'lat'</font><font color="4444FF">]</font>, <font color="4444FF">(</font><font color="#008000">' white &quot;'</font><font color="4444FF">+</font><font color="#2040a0">row</font><font color="4444FF">[</font><font color="#008000">'name'</font><font color="4444FF">]</font><font color="4444FF">+</font><font color="#008000">'&quot;'</font><font color="4444FF">)</font>
