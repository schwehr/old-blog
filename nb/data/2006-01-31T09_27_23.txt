and got it to compile.  Here is what I changed in gsf.h:
<pre>
<font color="0000ff"><strong>#ifndef __APPLE__</strong></font>
<font color="0000ff"><strong>#if (!defined (_STRUCT_TIMESPEC_) &amp;&amp; !defined (_TIMESPEC_T) &amp;&amp; !defined (_STRUCT_TIMESPEC) &amp;&amp; !defined (_SYS_TIMESPEC_H) &amp;&amp; !defined (__timespec_defined))</strong></font>
<font color="0000ff"><strong>#define _STRUCT_TIMESPEC_</strong></font>
<font color="0000ff"><strong>#define _TIMESPEC_T</strong></font>
<font color="0000ff"><strong>#define _STRUCT_TIMESPEC</strong></font>
