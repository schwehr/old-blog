Here is a dirt simple data logger for connecting to a port and logging
AIS data from a base station.  Your port number and IP address will
be different depending on vendor and configuration issues.
<pre>
<font color="#444444">#!/usr/bin/env python</font>
<strong>import</strong> <font color="#2040a0">socket</font>,<font color="#2040a0">select</font>
<strong>import</strong> <font color="#2040a0">sys</font>
<strong>import</strong> <font color="#2040a0">time</font>
<font color="#2040a0">s</font> <font color="4444FF">=</font> <font color="#2040a0">socket</font>.<font color="#2040a0">socket</font><font color="4444FF">(</font><font color="#2040a0">socket</font>.<font color="#2040a0">AF_INET</font>, <font color="#2040a0">socket</font>.<font color="#2040a0">SOCK_STREAM</font><font color="4444FF">)</font>
<font color="#2040a0">s</font>.<font color="#2040a0">connect</font><font color="4444FF">(</font><font color="4444FF">(</font><font color="#008000">'10.1.1.29'</font>, <font color="#FF0000">5505</font><font color="4444FF">)</font><font color="4444FF">)</font>
<font color="#2040a0">buf</font><font color="4444FF">=</font><font color="#008000">''</font>
<strong>while</strong> <font color="#2040a0">True</font><font color="4444FF">:</font>
    <font color="#2040a0">readersready</font>,<font color="#2040a0">outputready</font>,<font color="#2040a0">exceptready</font> <font color="4444FF">=</font> <font color="#2040a0">select</font>.<font color="#2040a0">select</font><font color="4444FF">(</font><font color="4444FF">[</font><font color="#2040a0">s</font><font color="4444FF">]</font>,<font color="4444FF">[</font><font color="4444FF">]</font>,<font color="4444FF">[</font><font color="4444FF">]</font>,.<font color="#FF0000">1</font><font color="4444FF">)</font>
    <strong>for</strong> <font color="#2040a0">sock</font> <strong>in</strong> <font color="#2040a0">readersready</font><font color="4444FF">:</font>
        <font color="#2040a0">data</font> <font color="4444FF">=</font> <font color="#2040a0">sock</font>.<font color="#2040a0">recv</font><font color="4444FF">(</font><font color="#FF0000">100</font><font color="4444FF">)</font>
        <font color="#2040a0">buf</font> <font color="4444FF">+</font><font color="4444FF">=</font> <font color="#2040a0">data</font>
        <font color="#2040a0">newline</font><font color="4444FF">=</font><font color="#2040a0">buf</font>.<font color="#2040a0">find</font><font color="4444FF">(</font><font color="#008000">'\n'</font><font color="4444FF">)</font>
        <strong>if</strong> <font color="4444FF">-</font><font color="#FF0000">1</font><font color="4444FF">!</font><font color="4444FF">=</font><font color="#2040a0">newline</font><font color="4444FF">:</font>
            <font color="#2040a0">fields</font> <font color="4444FF">=</font> <font color="#2040a0">buf</font>.<font color="#2040a0">split</font><font color="4444FF">(</font><font color="#008000">'\n'</font><font color="4444FF">)</font>
            <strong>print</strong> <font color="#2040a0">fields</font><font color="4444FF">[</font><font color="#FF0000">0</font><font color="4444FF">]</font>.<font color="#2040a0">strip</font><font color="4444FF">(</font><font color="4444FF">)</font><font color="4444FF">+</font><font color="#008000">','</font><font color="4444FF">+</font><font color="#2040a0">str</font><font color="4444FF">(</font><font color="#2040a0">time</font>.<font color="#2040a0">time</font><font color="4444FF">(</font><font color="4444FF">)</font><font color="4444FF">)</font>
            <strong>if</strong> <font color="#2040a0">len</font><font color="4444FF">(</font><font color="#2040a0">fields</font><font color="4444FF">)</font><font color="4444FF">&gt;</font><font color="#FF0000">1</font><font color="4444FF">:</font>
                <font color="#2040a0">buf</font><font color="4444FF">=</font><font color="#008000">''</font><font color="4444FF">+</font><font color="#2040a0">buf</font><font color="4444FF">[</font><font color="#2040a0">newline</font><font color="4444FF">+</font><font color="#FF0000">1</font><font color="4444FF">:</font><font color="4444FF">]</font>
            <strong>else</strong><font color="4444FF">:</font>
                <font color="#2040a0">buf</font><font color="4444FF">=</font><font color="#008000">''</font>
</pre>
