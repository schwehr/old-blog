This is as basic as it gets.  I wanted to try out
boost::dynamic_bitset.  It seems to work okay!  No idea on speed or
long term code development issues, but this is a start.
<pre>
<font color="0000ff"><strong>#include <font color="#008000">&lt;iostream&gt;</font></strong></font>
<font color="0000ff"><strong>#include <font color="#008000">&lt;boost/dynamic_bitset.hpp&gt;</font></strong></font>
<strong>using</strong> <strong>namespace</strong> <font color="#2040a0">boost</font><font color="4444FF">;</font>
<strong>int</strong> <font color="#2040a0">main</font> <font color="4444FF">(</font><strong>int</strong>, <strong>char</strong><font color="4444FF">*</font><font color="4444FF">[</font><font color="4444FF">]</font><font color="4444FF">)</font> <font color="4444FF"><strong>{</strong></font>
    <strong>const</strong> <strong>char</strong> <font color="#2040a0">c</font><font color="4444FF">=</font><font color="#008000">'1'</font><font color="4444FF">;</font>
    <strong>unsigned</strong> <strong>char</strong> <font color="#2040a0">val</font> <font color="4444FF">=</font> <font color="#2040a0">c</font> <font color="4444FF">-</font> <font color="#FF0000">48</font><font color="4444FF">;</font>
    <strong>if</strong> <font color="4444FF">(</font><font color="#2040a0">val</font><font color="4444FF">&gt;</font><font color="4444FF">=</font><font color="#FF0000">40</font><font color="4444FF">)</font> <font color="#2040a0">val</font> <font color="4444FF">-</font><font color="4444FF">=</font><font color="#FF0000">8</font><font color="4444FF">;</font>
    <strong>const</strong> <font color="#2040a0">dynamic_bitset</font><font color="4444FF">&lt;</font><font color="4444FF">&gt;</font> <font color="#2040a0">bits</font><font color="4444FF">(</font><font color="#FF0000">6</font>,<font color="#2040a0">val</font><font color="4444FF">)</font><font color="4444FF">;</font>
    <font color="#2040a0">std</font><font color="4444FF">:</font><font color="4444FF">:</font><font color="#2040a0">cout</font> <font color="4444FF">&lt;</font><font color="4444FF">&lt;</font> <font color="#008000">&quot;bits: &quot;</font> <font color="4444FF">&lt;</font><font color="4444FF">&lt;</font> <font color="#2040a0">bits</font> <font color="4444FF">&lt;</font><font color="4444FF">&lt;</font> <font color="#2040a0">std</font><font color="4444FF">:</font><font color="4444FF">:</font><font color="#2040a0">endl</font><font color="4444FF">;</font>
    <strong>return</strong> <font color="#FF0000">0</font><font color="4444FF">;</font>
<font color="4444FF"><strong>}</strong></font>
</pre>
And running it:
<pre>
# make
g++ -o cais cais.C -Wall -g -I/sw/include
# ./cais 
bits: 000001
</pre>
