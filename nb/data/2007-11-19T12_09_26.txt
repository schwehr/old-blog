On the JavaScript front: <a
href="http://workingrhino.blogspot.com/2007/10/getting-started.html">putting
rhino to work</a> gives a tip on how to get a readline like interface
for a javascript interpreter.
<pre>
<font color="0000ff"><strong>#!/bin/bash</strong></font>
<font color="#2040a0">dir</font>=$(dirname <font color="#2040a0">$0</font>)
export <font color="#2040a0">CLASSPATH</font>=${CLASSPATH}${CLASSPATH:+:}${dir}/rhino/\*
echo using <font color="#2040a0">CLASSPATH</font>=$(printenv CLASSPATH) &gt; /dev/stderr
[ <font color="#008000">&quot;<font color="#2040a0">$1</font>&quot;</font> = <font color="#008000">&quot;-f&quot;</font> ] || { <font color="#2040a0">RLWRAP</font>=<font color="#008000">&quot;rlwrap -C js&quot;</font> ;}
exec <font color="#2040a0">$RLWRAP</font> java ${JAVAOPTS} \
org.mozilla.javascript.tools.shell.Main -strict <font color="#008000">&quot;$@&quot;</font>
